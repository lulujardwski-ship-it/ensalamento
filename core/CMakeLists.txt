cmake_minimum_required(VERSION 3.16)
project(ensalamento_core LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(nlohmann_json 3.10 REQUIRED)
add_executable(ensalamento-core main.cpp)
target_link_libraries(ensalamento-core PRIVATE nlohmann_json::nlohmann_json)
if(MSVC)
  target_compile_options(ensalamento-core PRIVATE /W4 /utf-8)
else()
  target_compile_options(ensalamento-core PRIVATE -Wall -Wextra -Wpedantic)
endif()
